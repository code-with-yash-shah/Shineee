// -------------------- REGISTER --------------------
void register_user() {
    char username[MAX_USER_LEN + 1];
    char password[MAX_PASS_LEN + 1];
    char password_confirm[MAX_PASS_LEN + 1];

    printf("\n=== REGISTER ===\n");

    while (1) {
        printf("Enter username (no space): ");
        if (scanf("%50s", username) != 1) {
            printf("Input error. Try again.\n");
            clear_input_buffer();
            continue;
        }
        clear_input_buffer();

        if (strcmp(username, "0") == 0) {
            printf("Registration cancelled.\n");
            return;
        }

        if (username_exists(username)) {
            printf("Username already exists. Please choose another.\n");
            continue;
        }

        break;
    }

    while (1) {
        printf("Enter password (min 4 chars): ");
        if (scanf("%50s", password) != 1) {
            printf("Input error. Try again.\n");
            clear_input_buffer();
            continue;
        }
        clear_input_buffer();

        if (strlen(password) < 4) {
            printf("Password too short. Try again.\n");
            continue;
        }

        printf("Confirm password: ");
        scanf("%50s", password_confirm);
        clear_input_buffer();

        if (strcmp(password, password_confirm) != 0) {
            printf("Passwords do not match.\n");
            continue;
        }

        break;
    }

    if (save_user_to_file(username, password)) {
        printf("Registration successful! You can now login.\n");
    } else {
        printf("Error saving user.\n");
    }
}


// -------------------- LOGIN --------------------
void login_user() {
    char username[MAX_USER_LEN + 1];
    char password[MAX_PASS_LEN + 1];
    char stored_password[MAX_PASS_LEN + 1];

    printf("\n=== LOGIN ===\n");
    printf("Enter username: ");
    scanf("%50s", username);
    clear_input_buffer();

    int found = find_password_by_username(username, stored_password);
    if (found <= 0) {
        printf("User not found. Please register first.\n");
        return;
    }

    printf("Enter password: ");
    scanf("%50s", password);
    clear_input_buffer();

    if (strcmp(password, stored_password) == 0) {
        printf("\nLogin successful! Welcome, %s.\n", username);
        buy_rent();
    } else {
        printf("Incorrect password.\n");
    }
}

// -------------------- USER FILE FUNCTIONS --------------------
int username_exists(const char *username) {
    
    FILE *fp = fopen(USERS_FILE, "r");
    if (!fp) return 0;

    while (fscanf(fp, "%49[^|]|%49s\n", file_username, file_password) == 2) {
        if (strcmp(file_username, username) == 0) {
            fclose(fp);
            return 1;
        }
    }
    fclose(fp);
    return 0;
}

int save_user_to_file(const char *username, const char *password) {
    FILE *fp = fopen(USERS_FILE, "a");
    if (!fp) return 0;
    fprintf(fp, "%s|%s\n", username, password);
    fclose(fp);
    return 1;
}

int find_password_by_username(const char *username, char *out_password) {
    FILE *fp = fopen("users.txt", "r");
    if (!fp) return -1;

    while (fscanf(fp, "%49[^|]|%49s\n", file_username, file_password) == 2) {
        if (strcmp(file_username, username) == 0) {
            strcpy(out_password, file_password);
            fclose(fp);
            return 1;
        }
    }
    fclose(fp);
    return 0;
}

// -------------------- VIEW HISTORY --------------------
void view_history() {
    FILE *fp_purchase, *fp_rent;
    char line[300];
    int choice;

    while (1) {
        printf("\n=== VIEW HISTORY MENU ===\n");
        printf("1) View Purchase History\n");
        printf("2) View Rental History\n");
        printf("3) Return to Main Menu\n");
        printf("----------------------------\n");
        printf("Enter your choice: ");
        scanf("%d", &choice);

        if (choice == 3) {
            printf("Returning to main menu...\n");
            break;
        }

        if (choice == 1) {
            fp_purchase = fopen("purchases.txt", "r");
            if (!fp_purchase) {
                printf("\nNo purchase records found yet.\n");
                continue;
            }
            printf("\n=== PURCHASE HISTORY ===\n");
            int count = 0;
            while (fgets(line, sizeof(line), fp_purchase))
                printf("%d) %s", ++count, line);
            fclose(fp_purchase);
        } 
        else if (choice == 2) {
            fp_rent = fopen("rentals.txt", "r");
            if (!fp_rent) {
                printf("\nNo rental records found yet.\n");
                continue;
            }
            printf("\n=== RENTAL HISTORY ===\n");
            int count = 0;
            while (fgets(line, sizeof(line), fp_rent))
                printf("%d) %s", ++count, line);
            fclose(fp_rent);
        } 
        else {
            printf("Invalid choice. Try again.\n");
        }
    }
}


void clear_input_buffer() {
    int c;
    while ((c = getchar()) != '\n' && c != EOF);
}
